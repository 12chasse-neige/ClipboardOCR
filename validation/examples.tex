\section*{Energy and linear algebra}
1. Let $x_i=\alpha_i+\beta_i^2$ for $i=1,2,3$. The total energy is
\begin{equation}
 E = mc^2 + \frac{1}{2}mv^2 .
\end{equation}
2. For a matrix $A$ and vector $\mathbf{x}$, we have
\begin{equation}
 A\mathbf{x}=\begin{pmatrix}1&2\\3&4\end{pmatrix}\begin{pmatrix}x_1\\x_2\end{pmatrix}.
\end{equation}
\begin{itemize}
\item Preserve subscripts $a_{ij}$ and the symbol $\hbar$.
\item The fraction is $\frac{a+b}{c+d}$, with $c+d\ne0$.
\end{itemize}

\section*{数学与物理：基本关系}
1. 设 $x_i=\alpha_i+\beta_i^2$，其中 $i=1,2,3$。能量满足
\begin{equation}
 E=mc^2+\frac{1}{2}mv^2 .
\end{equation}
2. 矩阵与向量的乘积为
\begin{equation}
 A\mathbf{x}=\begin{pmatrix}1&2\\3&4\end{pmatrix}\begin{pmatrix}x_1\\x_2\end{pmatrix}.
\end{equation}
\begin{itemize}
\item 保留下标 $a_{ij}$、希腊字母 $\alpha$ 和符号 $\hbar$。
\item 当 $c+d\ne0$ 时，分式为 $\frac{a+b}{c+d}$。
\end{itemize}
